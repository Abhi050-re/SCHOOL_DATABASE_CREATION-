CREATE TABLE Dim_Department						-- CREATE DIMENSION TABLE WITH A PRIMARY KEY AND IDENTITY FOR ID'S(SURROGATE KEY)
(
	DepartmentKey INT IDENTITY(1,1) PRIMARY KEY,
	DepartmentID VARCHAR(50) UNIQUE,
	DepartmentName VARCHAR(50)
);

INSERT INTO                                           -- INSERT VALUES FROM THE ACTUAL TABLE TO THE DIMENSION TABLE
	Dim_Department(DepartmentID, DepartmentName)
	Select DISTINCT dept_id, dept
	From Teacher;

CREATE TABLE Dim_Teacher							-- CREATE ANOTHER DIMENSION TABLE 
(
	TeacherKey INT IDENTITY(1,1) PRIMARY KEY,
	TeacherID INT UNIQUE,
	TeacherName VARCHAR(50)
);

INSERT INTO
	Dim_Teacher(TeacherID, TeacherName)

CREATE TABLE FACT_Teacher            -- CREATE FACT TABLE FOR FACT_TEACHER WITH FORIEGN KEY LINKING TO DIMENSION TABLE, HERE YOU CAN STORE AND ACCESS VALUES
(	
	TeacherFactKey INT Identity(1,1) Primary key,    -- HERE WE HAVE LINKED TeacherKey and DepartmentKey from Dimension Tables for Teacher and Department 
	TeacherID INT UNIQUE,
	TeacherKey INT,
	Salary Decimal(12,2),
	Experience Decimal(12,2)

			Foreign Key (TeacherKey)						-- LINKING THEM TO DIM TABLE USING FORIEGN KEYS AND REFERENECING THEM
			REFERENCES Dim_Teacher(TeacherKey),             -- REFERENECING SHOWS WHAT THAT KEY REFERENCES TO

			Foreign Key (DepartmentKey)
			REFERENCES Dim_Department (DepartmentKey)
);
